# TODA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build TODA — a deferred-issues tracker: a stateless stdio MCP server (5 tools) over a plain `TODA.md` ledger file, per the approved spec at `docs/superpowers/specs/2026-10-27-toda-design.md`.

**Architecture:** `TODA.md` (+ `TODA-archive.md`) is the single source of truth. `src/store.ts` holds all format logic as pure functions (string in, string out); `src/io.ts` is the only module touching the filesystem; `src/server.ts` wires the five MCP tools; `src/main.ts` is the stdio entry point. Tests run the tools end-to-end over the SDK in-memory transport.

**Tech Stack:** Node ≥ 20 (host has 24.14), TypeScript (strict, NodeNext), `@modelcontextprotocol/sdk` ^1, `zod` ^3, `vitest` ^3.

## Global Constraints

- Dependencies limited to: `@modelcontextprotocol/sdk` ^1.12.0, `zod` ^3.24.0 (pin major 3 — SDK compatibility), dev: `typescript` ^5.6, `vitest` ^3, `@types/node` ^22. Nothing else.
- Header lines, line grammar, statuses (`o`|`s`|`x`), and sections (`Open`/`Sleeping`/`Done`) are EXACTLY as spec §6 — reproduce verbatim, do not reword the header comments.
- Segment delimiter is exactly ` / ` (space-slash-space); writers escape it inside user text to ` ; `; single `/` never splits.
- Responses are raw canonical lines; errors are one line: cause + next action. No JSON envelopes, no prose padding.
- No network, no subprocess calls, no git/GitHub resolution anywhere.
- TypeScript strict mode; ESM only (`"type": "module"`); relative TS imports use `.js` extensions (NodeNext).
- Host is Windows (Git Bash); tests use `os.tmpdir()` for scratch dirs.
- Spec is the authority for any question this plan doesn't answer: `docs/superpowers/specs/2026-10-27-toda-design.md`.

```typescript
export const STATUSES = ['o', 's', 'x'] as const;
export type Status = (typeof STATUSES)[number];

export const SECTION_BY_STATUS: Record<Status, string> = { o: 'Open', s: 'Sleeping', x: 'Done' };

export interface Item {
  id: string;
  status: Status;
  opened: string;
  where: string;
  what: string;
  why: string;
  revive: string;
  refs: string[];
  done?: string;
}

export interface ParsedItem extends Item {
  raw: string;
}

export interface ParsedDoc {
  items: ParsedItem[];
  unparsed: number;
}

const ITEM_RE = /^(D-\d+)\s+([osx])\s+(\d{4}-\d{2}-\d{2})\s+(\S+)\s+(.+)$/;

function isNoise(line: string): boolean {
  const t = line.trim();
  return t === '' || t.startsWith('#') || t.startsWith('<!--');
}

export function cleanText(s: string): string {
  return s.replace(/\r?\n/g, ' ').replace(/ \/ /g, ' ; ').trim();
}

export function cleanWhere(s: string): string {
  return cleanText(s).replace(/\s+/g, '-') || '-';
}

export function parse(content: string): ParsedDoc {
  const items: ParsedItem[] = [];
  let unparsed = 0;
  for (const line of content.split('\n')) {
    const m = ITEM_RE.exec(line);
    if (m) items.push({ ...itemFromMatch(m), raw: line });
    else if (!isNoise(line)) unparsed++;
  }
  return { items, unparsed };
}

function itemFromMatch(m: RegExpExecArray): Item {
  const [, id, status, opened, where, rest] = m;
  const item: Item = { id, status: status as Status, opened, where, what: '', why: '', revive: '', refs: [] };
  const untyped: string[] = [];
  for (const seg of rest.split(' / ')) {
    if (seg.startsWith('revive:')) item.revive = seg.slice('revive:'.length).trim();
    else if (seg.startsWith('ref:')) item.refs = seg.slice('ref:'.length).trim().split(/\s+/).filter(Boolean);
    else if (seg.startsWith('done:')) item.done = seg.slice('done:'.length).trim();
    else untyped.push(seg);
  }
  item.what = untyped.shift() ?? '';
  item.why = untyped.join(' / ');
  return item;
}

export function serializeItem(item: Item): string {
  const segs: string[] = [cleanText(item.what), cleanText(item.why)];
  if (item.revive !== '') segs.push(`revive: ${cleanText(item.revive)}`);
  if (item.refs.length > 0) segs.push(`ref: ${item.refs.map(cleanText).join(' ')}`);
  if (item.done !== undefined && item.done !== '') segs.push(`done: ${cleanText(item.done)}`);
  return `${item.id}  ${item.status}  ${item.opened}  ${cleanWhere(item.where)}  ${segs.join(' / ')}`;
}
```

```typescript
export function nextId(...contents: (string | null | undefined)[]): string {
  let max = 0;
  for (const c of contents) {
    if (!c) continue;
    for (const m of c.matchAll(/^D-(\d+)\s/gm)) max = Math.max(max, Number(m[1]));
  }
  return `D-${String(max + 1).padStart(3, '0')}`;
}
```

```typescript
export function insertIntoSection(content: string, section: string, line: string): string {
  const lines = content.split('\n');
  const h = lines.findIndex((l) => l.trim() === `## ${section}`);
  if (h === -1) {
    while (lines.length > 0 && lines[lines.length - 1] === '') lines.pop();
    lines.push('', `## ${section}`, line);
    return lines.join('\n') + '\n';
  }
  let end = lines.length;
  for (let i = h + 1; i < lines.length; i++) {
    if (lines[i].trim().startsWith('## ')) {
      end = i;
      break;
    }
  }
  lines.splice(end, 0, line);
  return lines.join('\n');
}

export interface AddInput {
  what: string;
  why: string;
  revive: string;
  where?: string;
  status?: Status;
  refs?: string[];
}

export function addItem(content: string, input: AddInput, opened: string, id?: string): { content: string; item: Item } {
  const item: Item = {
    id: id ?? nextId(content),
    status: input.status ?? 'o',
    opened,
    where: cleanWhere(input.where ?? '-'),
    what: cleanText(input.what),
    why: cleanText(input.why),
    revive: cleanText(input.revive),
    refs: input.refs ?? [],
  };
  return { content: insertIntoSection(content, SECTION_BY_STATUS[item.status], serializeItem(item)), item };
}
```

```typescript
export interface SetInput {
  status: Status;
  note?: string;
  refs?: string[];
}

export function setStatus(content: string, id: string, input: SetInput): { content: string; item: Item } | null {
  const found = parse(content).items.find((i) => i.id === id);
  if (!found) return null;
  const updated: Item = {
    ...found,
    status: input.status,
    refs: [...new Set([...found.refs, ...(input.refs ?? [])])],
    done: input.status === 'x' ? (input.note ?? found.done ?? '') : found.done,
  };
  const lines = content.split('\n');
  lines.splice(lines.indexOf(found.raw), 1);
  const out = insertIntoSection(lines.join('\n'), SECTION_BY_STATUS[updated.status], serializeItem(updated));
  return { content: out, item: updated };
```

```typescript
export function prune(
  content: string,
  archive: string | null,
  cutoff: string,
): { content: string; archive: string; moved: number } {
  const moving = new Set(
    parse(content)
      .items.filter((i) => i.status === 'x' && i.opened < cutoff)
      .map((i) => i.raw),
  );
  if (moving.size === 0) return { content, archive: archive ?? '', moved: 0 };
  const hotLines = content.split('\n').filter((l) => !moving.has(l));
  let arch = archive !== null && archive.trim() !== '' ? archive : emptyFile();
  for (const line of moving) arch = insertIntoSection(arch, 'Done', line);
  return { content: hotLines.join('\n'), archive: arch, moved: moving.size };
}
```

```typescript
export interface ListFilter {
  statuses?: Status[];
  keyword?: string;
  ref?: string;
  includeArchive?: boolean;
}

export function listItems(
  content: string,
  filter: ListFilter = {},
  archive?: string | null,
): { lines: string[]; unparsed: number } {
  const statuses = filter.statuses ?? ['o', 's'];
  const kw = filter.keyword?.toLowerCase();
  const lines: string[] = [];
  let unparsed = 0;
  const collect = (c: string) => {
    const doc = parse(c);
    unparsed += doc.unparsed;
    for (const it of doc.items) {
      if (!statuses.includes(it.status)) continue;
      if (kw && !it.raw.toLowerCase().includes(kw)) continue;
      if (filter.ref && !it.refs.some((r) => r.includes(filter.ref!))) continue;
      lines.push(it.raw);
    }
  };
  collect(content);
  if (filter.includeArchive && archive) collect(archive);
  return { lines, unparsed };
}
```

```typescript
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import * as store from './store.js';
import * as io from './io.js';

export const PROTOCOL_SNIPPET =
  '**TODA rule:** When we defer an issue or idea, call `toda_add` in the same turn — never "at the end of the session". ' +
  'When starting a task, call `toda_list` and mention open items whose `revive:` trigger matches the task. ' +
  'When a fix lands, call `toda_set` with status `x` in the same turn.';

function text(t: string) {
  return { content: [{ type: 'text' as const, text: t }] };
}

function err(t: string) {
  return { ...text(t), isError: true as const };
}

function refs(s: string | undefined): string[] {
  return s ? s.split(/\s+/).filter(Boolean) : [];
}

export function buildServer(): McpServer {

    if (existing === null) io.writeFile(p, store.emptyFile());
    return text(`${existing === null ? 'created' : 'already exists'}: ${p}\n\n${PROTOCOL_SNIPPET}`);
  });

  server.tool('toda_add', 'Append a deferred item to TODA.md (auto-creates the file)', {
    what: z.string().describe('One-line description of the issue'),
    why_deferred: z.string().describe('Why it is not being fixed now'),
    revive_when: z.string().describe('Trigger condition to revisit'),
    where: z.string().optional().describe('Location token like src/x.ts:42 (default -)'),
    status: z.enum(['o', 's', 'x']).optional().describe('Defaults to o (open)'),
    ref: z.string().optional().describe('Space-separated refs: commit shas, gh:#123, tag names'),
  }, async (a) => {
    const p = io.todaPath();
    const hot = io.readIfExists(p) ?? store.emptyFile();
    const archive = io.readIfExists(io.archivePath());
    const { content, item } = store.addItem(
      hot,
      {
        what: a.what,
        why: a.why_deferred,
        revive: a.revive_when,
        where: a.where,
        status: a.status,
        refs: refs(a.ref),
      },
      store.today(),
      store.nextId(hot, archive),
    );
    io.writeFile(p, content);
    return text(`added ${item.id}\n${store.serializeItem(item)}`);
  });

  return server;
}
```

```typescript
  server.tool('toda_set', 'Update an item status (moves it between sections)', {
    id: z.string().describe('Item id, e.g. D-017'),
    status: z.enum(['o', 's', 'x']),
    note: z.string().optional().describe('Closure note (used when status=x), e.g. the fix commit'),
    ref: z.string().optional().describe('Space-separated refs to merge into the item'),
  }, async (a) => {
    const p = io.todaPath();
    const hot = io.readIfExists(p);
    if (hot === null) return err('no TODA.md here; call toda_init');
    const res = store.setStatus(hot, a.id, { status: a.status, note: a.note, refs: refs(a.ref) });
    if (res === null) return err(`unknown id ${a.id}; call toda_list to see current ids`);
    io.writeFile(p, res.content);
    return text(store.serializeItem(res.item));
  });

  server.tool('toda_prune', 'Move done items older than N days into TODA-archive.md', {
    older_than_days: z.number().int().positive().optional().describe('Default 30'),
  }, async (a) => {
    const p = io.todaPath();
    const hot = io.readIfExists(p);
    if (hot === null) return err('no TODA.md here; call toda_init');
    const days = a.older_than_days ?? 30;
    const cutoff = new Date(Date.now() - days * 86400000).toISOString().slice(0, 10);
    const ap = io.archivePath();
    const res = store.prune(hot, io.readIfExists(ap), cutoff);
    if (res.moved > 0) {
      io.writeFile(p, res.content);
      io.writeFile(ap, res.archive);
    }
    return text(`pruned ${res.moved} item(s) to ${ap}`);
  });
```

```markdown
Tracks issues deliberately put aside during LLM-assisted work ("not a priority, later"),
so they stop getting lost between sessions. One plain-text ledger per project
```
